# Builds the overlay and two independent, optional integration targets.
# Base platform/gpu source lists never compile an ImGui backend or adapter.
if(NOT SOMNIUM_ENABLE_DEBUG_UI)
    return()
endif()
foreach(module platform gpu)
    set(integration somnium_${module}_imgui)
    set(source_root "${PROJECT_SOURCE_DIR}/engine/${module}")
    if(module STREQUAL "platform")
        set(backend imgui_impl_sdl3)
    else()
        set(backend imgui_impl_sdlgpu3)
    endif()
    add_library(${integration} STATIC "${imgui_SOURCE_DIR}/backends/${backend}.cpp"
        "${source_root}/integration/adapter.cpp")
    target_include_directories(${integration} PUBLIC "${source_root}/integration/include"
        PRIVATE "${source_root}/private")
    target_include_directories(${integration} SYSTEM PRIVATE "${imgui_SOURCE_DIR}/backends")
    target_link_libraries(${integration} PRIVATE somnium::${module} somnium_imgui SDL3::SDL3)
    # Upstream backends retain their own warning policy; our adapters use project warnings.
    if(MSVC)
        target_compile_options(${integration} PRIVATE /GR- /EHs-c-)
        set_source_files_properties("${source_root}/integration/adapter.cpp" PROPERTIES COMPILE_OPTIONS "/W4;/permissive-;/utf-8;$<$<BOOL:${SOMNIUM_WARNINGS_AS_ERRORS}>:/WX>")
    else()
        target_compile_options(${integration} PRIVATE -fno-exceptions -fno-rtti)
        set_source_files_properties("${source_root}/integration/adapter.cpp" PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wpedantic;$<$<BOOL:${SOMNIUM_WARNINGS_AS_ERRORS}>:-Werror>")
    endif()
    somnium_sanitizers(${integration})
endforeach()
add_library(somnium_debug_ui STATIC src/overlay.cpp)
add_library(somnium::debug_ui ALIAS somnium_debug_ui)
target_include_directories(somnium_debug_ui PUBLIC include)
target_link_libraries(somnium_debug_ui PUBLIC somnium::core PRIVATE
    somnium_platform_imgui somnium_gpu_imgui somnium_imgui)
somnium_configure_target(somnium_debug_ui)
if(BUILD_TESTING)
    somnium_add_test(somnium_debug_ui_tests tests/frame_times_tests.cpp debug_ui)
    target_include_directories(somnium_debug_ui_tests PRIVATE src)
endif()
